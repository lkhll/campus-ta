### 2.3 写什么

**文件 1：`llm_client.py`**

```python
"""DeepSeek 客户端：流式、重试、token 统计。"""
import os
import time
from dataclasses import dataclass

from dotenv import load_dotenv
from openai import OpenAI, APITimeoutError, RateLimitError, APIError

load_dotenv()

MODEL = "deepseek-flash"
BASE_URL = "https://api.deepseek.com"

client = OpenAI(
    api_key=os.getenv("DEEPSEEK_API_KEY", ""),
    base_url=BASE_URL,
    timeout=60.0,
    max_retries=2,
)


@dataclass
class Usage:
    prompt_tokens: int = 0
    completion_tokens: int = 0
    total_tokens: int = 0


def chat(messages, temperature=0.7, max_tokens=1024):
    """非流式调用，返回 (文本, Usage)。"""
    last_err = None
    for attempt in range(3):
        try:
            resp = client.chat.completions.create(
                model=MODEL,
                messages=messages,
                temperature=temperature,
                max_tokens=max_tokens,
            )
            u = resp.usage
            return resp.choices[0].message.content, Usage(
                u.prompt_tokens, u.completion_tokens, u.total_tokens
            )
        except (APITimeoutError, RateLimitError, APIError) as e:
            last_err = e
            time.sleep(2 ** attempt)
    raise RuntimeError(f"调用失败：{last_err}")


def stream_chat(messages, usage: dict | None = None, **kwargs):
    """流式调用，逐段 yield 文本；传 usage 字典时回填 token。"""
    stream = client.chat.completions.create(
        model=MODEL,
        messages=messages,
        stream=True,
        stream_options={"include_usage": True},
        **kwargs,
    )
    for chunk in stream:
        if chunk.usage and usage is not None:
            usage["prompt_tokens"] = chunk.usage.prompt_tokens
            usage["completion_tokens"] = chunk.usage.completion_tokens
            usage["total_tokens"] = chunk.usage.total_tokens
        if chunk.choices and chunk.choices[0].delta.content:
            yield chunk.choices[0].delta.content
```

> 官方文档确认支持 `stream_options.include_usage`；如果个别版本报错，删掉那一行即可，最后一个 chunk 仍会带 usage。

**文件 2：`cli_chat.py`**

```python
"""命令行多轮对话：流式输出 + token 统计。"""
from llm_client import stream_chat

SYSTEM = "你是一个简洁、准确的中文助手。不确定时直接说不知道，不要编造。"


def main():
    messages = [{"role": "system", "content": SYSTEM}]
    total = 0
    print("输入 exit 退出，/clear 清空上下文。")
    while True:
        try:
            user = input("\n你> ").strip()
        except (EOFError, KeyboardInterrupt):
            print()
            break
        if user in {"exit", "quit"}:
            break
        if user == "/clear":
            messages = [{"role": "system", "content": SYSTEM}]
            total = 0
            print("上下文已清空")
            continue
        if not user:
            continue

        messages.append({"role": "user", "content": user})
        if len(messages) > 21:
            messages = messages[:1] + messages[-20:]  # 只保留最近 20 条

        print("AI> ", end="", flush=True)
        parts, usage = [], {}
        try:
            for piece in stream_chat(messages, usage=usage):
                print(piece, end="", flush=True)
                parts.append(piece)
        except Exception as e:
            print(f"\n[错误] {type(e).__name__}: {e}")
            messages.pop()
            continue

        answer = "".join(parts)
        messages.append({"role": "assistant", "content": answer})
        total += usage.get("total_tokens", 0)
        print(f"\n[tokens] 本次 {usage.get('total_tokens', 0)}，累计 {total}")


if __name__ == "__main__":
    main()
```

**文件 2：`structured.py`**（结构化输出 + 校验 + 重试）

```python
import json

from pydantic import BaseModel, ValidationError

from llm_client import chat
from prompts import JD_EXTRACT_PROMPT


class JDSchema(BaseModel):
    title: str
    skills: list[str]
    min_years: int | None = None
    salary: str | None = None


def extract_jd(jd: str, retries: int = 2) -> JDSchema:
    prompt = JD_EXTRACT_PROMPT.format(jd=jd)
    for i in range(retries + 1):
        text, _ = chat(
            [{"role": "user", "content": prompt}],
            temperature=0,
        )
        text = text.strip()
        if text.startswith("```"):
            text = text.split("```")[1]
            if text.startswith("json"):
                text = text[4:]
        try:
            return JDSchema.model_validate_json(text.strip())
        except (ValidationError, json.JSONDecodeError):
            if i == retries:
                raise
    raise RuntimeError("unreachable")


if __name__ == "__main__":
    jd = "招 AI 应用开发实习生，要求熟悉 Python、LangChain，了解 RAG，每周 5 天，150-180 元/天。"
    print(extract_jd(jd))
```

**文件 3：`tools.py`**

```python
import sqlite3

from pydantic import BaseModel, Field

_conn = sqlite3.connect(":memory:", check_same_thread=False)
_conn.execute("CREATE TABLE scores (name TEXT, course TEXT, score INTEGER)")
_conn.executemany(
    "INSERT INTO scores VALUES (?, ?, ?)",
    [("张三", "高数", 88), ("张三", "英语", 76), ("李四", "高数", 92)],
)
_conn.commit()


class WeatherArgs(BaseModel):
    city: str = Field(description="城市名，例如 杭州")


class ScoreArgs(BaseModel):
    name: str = Field(description="学生姓名")


def get_weather(city: str) -> dict:
    fake = {"杭州": "多云 18-24℃", "郑州": "晴 10-20℃"}
    return {"city": city, "weather": fake.get(city, "暂无数据")}


def query_score(name: str) -> dict:
    rows = _conn.execute(
        "SELECT course, score FROM scores WHERE name = ?", (name,)
    ).fetchall()
    return {"name": name, "scores": [{"course": c, "score": s} for c, s in rows]}


TOOLS = [
    {
        "type": "function",
        "function": {
            "name": "get_weather",
            "description": "查询指定城市今天的天气。用户问天气、气温、下雨时调用。",
            "parameters": WeatherArgs.model_json_schema(),
        },
    },
    {
        "type": "function",
        "function": {
            "name": "query_score",
            "description": "按学生姓名查询各科成绩。用户问成绩、分数时调用。",
            "parameters": ScoreArgs.model_json_schema(),
        },
    },
]

REGISTRY = {
    "get_weather": (WeatherArgs, get_weather),
    "query_score": (ScoreArgs, query_score),
}


def dispatch(name: str, args: dict) -> dict:
    if name not in REGISTRY:
        raise ValueError(f"未知工具：{name}")
    model, fn = REGISTRY[name]
    parsed = model(**args)
    return fn(**parsed.model_dump())
```

**文件 4：`agent_loop.py`**

```python
import json
import os

from dotenv import load_dotenv
from openai import OpenAI

from tools import TOOLS, dispatch

load_dotenv()

client = OpenAI(
    api_key=os.getenv("DEEPSEEK_API_KEY"),
    base_url="https://api.deepseek.com",
)
MODEL = "deepseek-flash"


def run_agent(messages, max_steps=5, verbose=True):
    for step in range(1, max_steps + 1):
        resp = client.chat.completions.create(
            model=MODEL,
            messages=messages,
            tools=TOOLS,
            tool_choice="auto",
        )
        msg = resp.choices[0].message
        messages.append(msg.model_dump(exclude_none=True))

        if not msg.tool_calls:
            return msg.content

        for tc in msg.tool_calls:
            name = tc.function.name
            args = {}
            try:
                args = json.loads(tc.function.arguments or "{}")
                result = dispatch(name, args)
            except Exception as e:
                result = {"error": f"{type(e).__name__}: {e}"}
            if verbose:
                print(f"[step {step}] {name}({args}) -> {result}")
            messages.append(
                {
                    "role": "tool",
                    "tool_call_id": tc.id,
                    "content": json.dumps(result, ensure_ascii=False),
                }
            )

    return "达到最大步数，任务未完成"


if __name__ == "__main__":
    msgs = [{"role": "system", "content": "你是校园助手，需要数据时调用工具。"}]
    print(run_agent(msgs + [{"role": "user", "content": "张三高数多少分？"}]))
```

### 3.5 写工具的关键

工具描述要写清三件事，模型才选得准：

1. **什么时候用**：“用户问天气、气温、下雨时调用”

3. **失败返回什么**：用 `{"error": "..."}`，不要抛异常给模型

### 4.6 SSE 视图

`chat/views.py`：

```python
import json

from django.http import JsonResponse, StreamingHttpResponse
from django.shortcuts import render
from django.views.decorators.csrf import csrf_exempt
from django.views.decorators.http import require_POST

from llm_client import stream_chat

from .models import Conversation, Message


def index(request):
    return render(request, "index.html")


@csrf_exempt
@require_POST
def chat_stream(request):
    data = json.loads(request.body or "{}")
    text = (data.get("message") or "").strip()
    conv_id = data.get("conversation_id")
    if not text:
        return JsonResponse({"error": "message 不能为空"}, status=400)

    conv = Conversation.objects.filter(id=conv_id).first() if conv_id else None
    if conv is None:
        conv = Conversation.objects.create()

    Message.objects.create(conversation=conv, role="user", content=text)
    history = [
        {"role": m.role, "content": m.content}
        for m in conv.messages.order_by("-created_at")[:50]
    ][::-1]

    def event_stream():
        parts, usage = [], {}
        try:
            yield f"data: {json.dumps({'type': 'meta', 'conversation_id': conv.id})}\n\n"
            for piece in stream_chat(history, usage=usage):
                parts.append(piece)
                yield (
                    "data: "
                    + json.dumps(
                        {"type": "delta", "text": piece}, ensure_ascii=False
                    )
                    + "\n\n"
                )
            yield f"data: {json.dumps({'type': 'done', 'usage': usage})}\n\n"
        except Exception as e:
            yield (
                "data: "
                + json.dumps(
                    {"type": "error", "message": str(e)}, ensure_ascii=False
                )
                + "\n\n"
            )
        finally:
            if parts:
                Message.objects.create(
                    conversation=conv, role="assistant", content="".join(parts)
                )

    resp = StreamingHttpResponse(event_stream(), content_type="text/event-stream")
    resp["Cache-Control"] = "no-cache"
    resp["X-Accel-Buffering"] = "no"
    return resp


def history(request):
    conv_id = request.GET.get("id")
    if not conv_id:
        return JsonResponse({"messages": []})
    msgs = Message.objects.filter(conversation_id=conv_id).order_by("created_at")
    return JsonResponse(
        {"messages": [{"role": m.role, "content": m.content} for m in msgs]}
    )
```

### 4.7 前端

新建 `templates/index.html`：

```html
<!doctype html>
<html lang="zh-CN">
<head>
  <meta charset="utf-8">
  <title>CampusTA</title>
  <style>
    body { font-family: system-ui, sans-serif; max-width: 720px; margin: 40px auto; padding: 0 16px; }
    #messages { border: 1px solid #ddd; border-radius: 8px; height: 480px; overflow-y: auto; padding: 12px; }
    .user { background: #e8f0fe; margin: 8px 0; padding: 8px 12px; border-radius: 8px; }
    .assistant { background: #f5f5f5; margin: 8px 0; padding: 8px 12px; border-radius: 8px; white-space: pre-wrap; }
    #input { width: 100%; height: 72px; margin-top: 12px; }
    button { margin-top: 8px; padding: 8px 24px; }
  </style>
</head>
<body>
  <h2>CampusTA</h2>
  <div id="messages"></div>
  <textarea id="input" placeholder="输入问题，Enter 发送，Shift+Enter 换行"></textarea>
  <button onclick="send()">发送</button>

  <script>
    let conversationId = localStorage.getItem("cid");
    const box = document.getElementById("messages");

    function bubble(role, text) {
      const div = document.createElement("div");
      div.className = role;
      div.textContent = text;
      box.appendChild(div);
      box.scrollTop = box.scrollHeight;
      return div;
    }

    async function loadHistory() {
      if (!conversationId) return;
      const r = await fetch(`/api/history/?id=${conversationId}`);
      const data = await r.json();
      data.messages.forEach(m => bubble(m.role, m.content));
    }

    async function send() {
      const input = document.getElementById("input");
      const text = input.value.trim();
      if (!text) return;
      input.value = "";
      bubble("user", text);
      const ai = bubble("assistant", "");

      const resp = await fetch("/api/chat/stream/", {
        method: "POST",
        headers: {"Content-Type": "application/json"},
        body: JSON.stringify({conversation_id: conversationId, message: text}),
      });
      if (!resp.ok) {
        ai.textContent = "请求失败：" + resp.status;
        return;
      }

      const reader = resp.body.getReader();
      const decoder = new TextDecoder();
      let buf = "";
      while (true) {
        const {done, value} = await reader.read();
        if (done) break;
        buf += decoder.decode(value, {stream: true});
        const events = buf.split("\n\n");
        buf = events.pop();
        for (const ev of events) {
          if (!ev.startsWith("data:")) continue;
          const payload = JSON.parse(ev.slice(5).trim());
          if (payload.type === "meta") {
            conversationId = payload.conversation_id;
            localStorage.setItem("cid", conversationId);
          } else if (payload.type === "delta") {
            ai.textContent += payload.text;
            box.scrollTop = box.scrollHeight;
          } else if (payload.type === "error") {
            ai.textContent += "\n[错误] " + payload.message;
          }
        }
      }
    }

    document.getElementById("input").addEventListener("keydown", e => {
      if (e.key === "Enter" && !e.shiftKey) {
        e.preventDefault();
        send();
      }
    });

    loadHistory();
  </script>
</body>
</html>
```

### 4.10 常见错误

| 现象 | 原因 | 处理 |
|---|---|---|
| 页面一直转圈 | 视图没加 `content_type="text/event-stream"` | 检查响应头 |
| 一次性全出来 | 中间有代理/缓冲 | 加 `X-Accel-Buffering: no` |
| `403 CSRF` | POST 没带 token | 开发期先 `@csrf_exempt`，上线前改回 |
| `no such table` | 没迁移 | `python manage.py migrate` |
| 刷新没历史 | localStorage 没存 | 看 `meta` 事件是否返回 `conversation_id` |
| 关页面报错 | 生成器被中断 | `finally` 里保存，异常吞掉 |
